// Requires an admin session. Saves productName, productModel and productSeo.
check that page contains "Dashboard"
click "Catalog"
click "Products"
check that page contains "Products"
click "Add New"
check that page contains "Product Name"
generate from template "TR Product #####", then enter into "Product Name" and save as "productName"
enter stored value "productName" into "Meta Tag Title"
click "Data"
generate from template "TRM#####", then enter into "Model" and save as "productModel"
enter "100" into "Quantity"
click "SEO"
generate from template "trseo#####", then enter into "keyword" and save as "productSeo"
click "Save"
check that page contains "Success: You have modified products!"
