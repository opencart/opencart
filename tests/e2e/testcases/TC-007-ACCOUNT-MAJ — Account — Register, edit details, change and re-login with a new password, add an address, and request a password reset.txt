// Full account lifecycle for one generated customer: each act verifies the previous one.

// --- Register (auto-logs the customer in) ---
customer-register-new
click "Continue"
check that page contains "Edit your account information"

// --- Edit the account information ---
click "Edit your account information"
check that page contains "First Name"
enter "Updated" into "First Name"
click "Continue"
check that page contains "Success: Your account has been successfully updated."

// Account saves are ajax and stay on the sub-page, so return via the form's Back button
click "Back"
check that page contains "Change your password"

// --- Change the password ---
click "Change your password"
check that page contains "Change Password"
generate by regex "[A-Z][a-z]{6}[0-9]{2}[!@#]", then enter into "Password" and save as "newPassword"
enter stored value "newPassword" into "Password Confirm"
click "Continue"
check that page contains "Success: Your password has been successfully updated."

// --- Prove the new password works, which the success banner alone does not ---
customer-logout
save stored value "newPassword" as "currentPassword"
customer-login

// --- Add an address book entry ---
click "Modify your address book entries"
check that page contains "Address Book"
click "New Address"
enter "Updated" into "First Name"
enter stored value "customerLastName" into "Last Name"
address-fill
click "Continue"
check that page contains "Your address has been successfully added"

// --- Request a password reset ---
customer-logout
click "My Account"
click "Login"
check that page contains "Returning Customer"
click "Forgotten Password"
check that page contains "Forgot Your Password?"
enter stored value "customerEmail" into "E-Mail Address"
click "Continue"
check that page contains "An email with a confirmation link has been sent to your email address"
